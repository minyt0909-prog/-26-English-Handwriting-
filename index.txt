<script>
  let currentUser = null;
  let currentGrade = 1;
  let currentChasiObj = null;

  // 캔버스 설정
  const canvas = document.getElementById('writeCanvas');
  const ctx = canvas.getContext('2d');
  let drawing = false;

  function resizeCanvas() {
    const box = document.getElementById('canvas-container');
    canvas.width = box.clientWidth;
    canvas.height = box.clientHeight;
    ctx.strokeStyle = '#1e293b';
    ctx.lineWidth = 3;
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';
  }
  window.addEventListener('resize', resizeCanvas);

  // S펜 및 터치 이벤트 핸들러 (팜레이젝션 포함)
  canvas.addEventListener('pointerdown', (e) => {
    drawing = true;
    ctx.beginPath();
    ctx.moveTo(e.offsetX, e.offsetY);
  });
  canvas.addEventListener('pointermove', (e) => {
    if (!drawing) return;
    ctx.lineTo(e.offsetX, e.offsetY);
    ctx.stroke();
  });
  canvas.addEventListener('pointerup', () => { drawing = false; });
  canvas.addEventListener('pointerleave', () => { drawing = false; });

  function clearCanvas() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
  }

  function handleLogin() {
    const grade = document.getElementById('grade-select').value;
    const id = document.getElementById('student-id').value.trim();
    const pw = document.getElementById('student-pw').value.trim();

    if (!id || !pw) {
      alert('학번과 비밀번호를 입력해주세요!');
      return;
    }

    currentGrade = parseInt(grade);
    currentUser = { id, grade: currentGrade };
    document.getElementById('user-display').innerText = `[${grade}학년] 학번: ${id}`;
    
    document.getElementById('login-section').classList.add('hidden');
    document.getElementById('app-section').classList.remove('hidden');

    initChasiDropdown();
    resizeCanvas();
  }

  function handleLogout() {
    currentUser = null;
    document.getElementById('app-section').classList.add('hidden');
    document.getElementById('login-section').classList.remove('hidden');
  }

  function initChasiDropdown() {
    const select = document.getElementById('chasi-select');
    select.innerHTML = '';
    const list = CURRICULUM_DATA[currentGrade];
    list.forEach(item => {
      const opt = document.createElement('option');
      opt.value = item.chasi;
      opt.innerText = `${item.chasi}차시: ${item.title}`;
      select.appendChild(opt);
    });
    changeChasi();
  }

  function changeChasi() {
    const chasiNum = parseInt(document.getElementById('chasi-select').value);
    currentChasiObj = CURRICULUM_DATA[currentGrade].find(item => item.chasi === chasiNum);
    document.getElementById('chasi-title').innerText = `${currentChasiObj.chasi}차시 연습목표: ${currentChasiObj.title}`;
    document.getElementById('guide-text').innerText = currentChasiObj.guideText;
    clearCanvas();
    document.getElementById('ai-result-box').classList.add('hidden');
  }

  // Gemini API를 활용한 비전 분석
  async function analyzeWithAI() {
    const dataUrl = canvas.toDataURL('image/png');
    const base64Data = dataUrl.split(',')[1];
    
    document.getElementById('ai-result-box').classList.remove('hidden');
    document.getElementById('ai-feedback-text').innerText = "🤖 AI가 학생의 4선지 필기를 정밀 분석 중입니다...";

    // 
    const apiKey = "AQ.Ab8RN6JQFLzkYHbIJJmoYTIiwnipUBBN_SQq9BUGHuNHt92Nuw";
    const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`;

    const promptText = `당신은 중학교 영어 교사입니다. 학생들이 4선지 노트에 쓴 영어 필기 이미지입니다. 
목표 문장/알파벳: "${currentChasiObj.guideText}"
이미지에 적힌 학생의 필기를 분석하여 아래 3가지 항목으로 친절하고 구체적으로 피드백해주세요:
1. AI가 읽어낸 학생의 글자 (OCR 인식 결과)
2. 4선지 기준선(Base-line, Ascender/Descender 선) 준수 여부 및 잘한 점
3. 개선을 위한 따뜻한 조언 1가지 (예: 'g가 q처럼 보여요', '베이스라인에 조금 더 밀착해 보세요')`;

    try {
      const response = await fetch(endpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          contents: [{
            parts: [
              { text: promptText },
              { inline_data: { mime_type: "image/png", data: base64Data } }
            ]
          }]
        })
      });

      const result = await response.json();
      const feedback = result.candidates[0].content.parts[0].text;
      document.getElementById('ai-feedback-text').innerHTML = feedback.replace(/\n/g, '<br>');

      // 파이어베이스 DB에 결과 저장
      const docId = `${currentUser.id}_g${currentGrade}_c${currentChasiObj.chasi}`;
      await window.setDoc(window.doc(window.db, "submissions", docId), {
        studentId: currentUser.id,
        grade: currentGrade,
        chasi: currentChasiObj.chasi,
        chasiTitle: currentChasiObj.title,
        imageData: dataUrl,
        feedback: feedback,
        timestamp: new Date().toISOString()
      });

    } catch (error) {
      console.error(error);
      document.getElementById('ai-feedback-text').innerText = "❌ AI 분석 중 오류가 발생했습니다. API 키를 확인해주세요.";
    }
  }

  // 교사 대시보드 보기
  async function openTeacherDashboard() {
    document.getElementById('login-section').classList.add('hidden');
    document.getElementById('teacher-section').classList.remove('hidden');
    
    const listDiv = document.getElementById('portfolio-list');
    listDiv.innerHTML = '<p>학생 제출 데이터를 불러오는 중...</p>';

    try {
      const querySnapshot = await window.getDocs(window.collection(window.db, "submissions"));
      if (querySnapshot.empty) {
        listDiv.innerHTML = '<p>아직 제출된 학생 데이터가 없습니다.</p>';
        return;
      }

      let html = '<table border="1" cellpadding="8" style="width:100%; border-collapse: collapse; text-align: left;">';
      html += '<tr><th>학번</th><th>학년</th><th>차시</th><th>제출 시간</th><th>AI 피드백 요약</th><th>필기 이미지</th></tr>';
      
      querySnapshot.forEach((doc) => {
        const d = doc.data();
        html += `<tr>
          <td><b>${d.studentId}</b></td>
          <td>${d.grade}학년</td>
          <td>${d.chasi}차시</td>
          <td>${d.timestamp.substring(5, 16)}</td>
          <td><small>${d.feedback.substring(0, 80)}...</small></td>
          <td><img src="${d.imageData}" width="120" style="border:1px solid #ccc; border-radius:4px;"></td>
        </tr>`;
      });
      html += '</table>';
      listDiv.innerHTML = html;
    } catch (e) {
      console.error(e);
      listDiv.innerHTML = '<p>데이터를 불러오는 중 오류가 발생했습니다. 파이어베이스 설정을 확인해주세요.</p>';
    }
  }

  function closeTeacherDashboard() {
    document.getElementById('teacher-section').classList.add('hidden');
    document.getElementById('login-section').classList.remove('hidden');
  }
</script>
